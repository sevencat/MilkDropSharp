#define dmin(a, b)	((a.x < b.x) ? a : b)
#define minkowski(v, m)	pow(dot(pow(v, v * 0. + m), v * 0. + 1.), 1. / m)
#define char(p,C)(((p.x<0.||p.x>1.||p.y<0.||p.y>1.))?vec4(0,0,0,1e5):textureGrad(iChannel3,p/16.+fract(vec2(C,15-C/16)/16.),dFdx(p/16.),dFdy(p/16.)))
	
vec2 pR(in vec2 p, float a) 
{
	p = cos(a) * p + sin(a) * vec2(p.y, -p.x);
	return vec2(p.xy);
}
vec3 crt( in vec3 p )
{
     float r =  .5 * 6.2831 / 6.;
     float pa = r - atan(p.x, p.y);
           pa += mod( pa,  -2. * r );
    return vec3(pR(p.xy, pa ), p.z);
}
vec2 f(vec3 p) 
{
	vec3 v_bb = p;
	v_bb.xz = pR(v_bb.xz, iTime);
	return dmin(vec2(minkowski(abs(abs(crt(abs(p) - vec3(1, .1, 3. + mod(floor(2. * iTime), 3.)))) - vec3(.7 + mod(floor(iTime), 3.), 4.2, 1.3)), 1.55) - 1.25, 3.), vec2(minkowski(abs(v_bb), 3.05) - (.9 + fract(.2 * floor(iTime))), 5));
}

vec3 N(in vec3 pos) 
{
	const vec3 eps = vec3(.001, 0, 0);
	return normalize(vec3(f(pos + eps.xyy).x - f(pos - eps.xyy).x, f(pos + eps.yxy).x - f(pos - eps.yxy).x, f(pos + eps.yyx).x - f(pos - eps.yyx).x));
}

void mainImage(out vec4 fragColor, in vec2 fragCoord) 
{
	vec2 d,
	     R = iResolution.xy,
	     st = -1. + 2. * fragCoord.xy / R, U = st;
         U.x*=R.x/R.y;
    float y = 0.;
    for(int i = 0;i<10;i++){
    y += (char(12.*vec2(U.x+1.75 , U.y+.85), int[](115,111,117, 110, 100, 32, 100, 101, 109, 111)[int(i)]).x);
    U.x-=.0549;
    }
     
    float h,aspct = max(R.x / R.y, R.y / R.x);
	vec3 rd, p, c,
	     ro = vec3(fract(iTime) , fract(iTime) , 7.),
	     gc = normalize( - ro),
	     u = normalize(cross(gc, vec3(0, 1, 0))),
	     v = normalize(cross(u, gc));
	u *= tan(1.2);
	v *= tan(1.2) / aspct;
	rd = normalize(gc + st.x * u + st.y * v);
	p = ro;
	c = (1.-y)*vec3(1);
	h = 1.;
	d = vec2(0., 0.);
	for (int i = 0; i < 32; ++i) 
    {
		p = ro + rd * h;
		d = f(p);
		if (abs(d.x) < .001) 
        {
			float fr = 1e-3;
			vec3 n = N(p);
			if (d.y == 3.) 
            {
				fr = sqrt(1. - dot(-rd, n));
				c *= fr;
				ro = p + n * .01;
				rd = reflect(rd, n);
				h = 0.;
			}

			if (d.y == 5.) 
            {
				fr = 1. - dot(-rd, n);
				c *= fr;
				ro = p + n * .001;
				rd = reflect(rd, n);
				h = 0.;
			}
		}
		h += d.x;
	}

   fragColor = vec4(c, 1);
}
